# Boots to the main menu and opens the Settings screen.
boot NORAND
ignore POOL OF RADIANCE V
# Verify the displayed build name.
ignore COMMIT  |VERSION  
# Record free memory, which changes with code size.
ignore FREE MEMORY|LARGEST BLOCK
# Record the data version maintained by DataPat and C_DATA.VERSION.
strip V\d+ +MERGED
wait RADIANCE
checkpoint main-menu
press g
checkpoint settings
