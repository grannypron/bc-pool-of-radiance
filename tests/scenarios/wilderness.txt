# Tests the central wilderness using save G. Open the wilderness map, force its second tile sheet to load through
# MapTiles.SecondSheet, walk a few steps, and record memory use afterward.
boot NORAND
ignore POOL OF RADIANCE V
wait RADIANCE
put saves/wild
# Load the slot directly by pressing its letter (#652).
press l g
wait ^(MOVE )?CAST VIEW ENCAMP 60
checkpoint wilderness
press up
wait ^(MOVE )?CAST VIEW ENCAMP 60
checkpoint north
press right
wait ^(MOVE )?CAST VIEW ENCAMP 60
checkpoint east
press down
wait ^(MOVE )?CAST VIEW ENCAMP 60
checkpoint south
# Record overlay loads and memory with STING Ctrl-Y. These values are reported but not compared. Also capture the heap
# and picture-memory lines to show what the wilderness leaves allocated for phase 6.
ignore OVERLAY LOADS
press ctrl-y
checkpoint overlay-loads
press ret ret
ignore HEAP FREE
checkpoint heap
press ret
checkpoint pictures
press ret
